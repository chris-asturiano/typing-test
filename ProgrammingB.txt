int main() {
    int result = 1;
    for (int i = 1; i <= 5; i++) {
        result = result * i;
    }
    printf("Result: %d", result);
    if (result > 50) {
        printf(" - Success");
    } else {
        printf(" - Too small");
    }
    result = result + 10;
    printf("\nAdjusted Result: %d", result);
    return 0;
}
